# Prueba de instrucciones tipo U: lui y auipc
lui   x1, 0x12345           # 0x00  x1 = 0x12345000
lui   x2, 0xfffff           # 0x04  x2 = 0xfffff000  (bit 31 en 1)
lui   x3, 0                 # 0x08  x3 = 0
auipc x4, 0                 # 0x0c  x4 = 0x0000000c  (pc de la propia auipc)
auipc x5, 1                 # 0x10  x5 = 0x00001010
auipc x6, 0xfffff           # 0x14  x6 = 0xfffff014  (pc - 4096, da la vuelta)
lui   x7, 0x80000           # 0x18  x7 = 0x80000000
addi  x7, x7, -1            # 0x1c  x7 = 0x7fffffff  (lui + addi = constante de 32 bits)
lui   x8, 0x12345           # 0x20
addi  x8, x8, 0x678         # 0x24  x8 = 0x12345678
lui   x9, 0x12345           # 0x28
addi  x9, x9, -0x800        # 0x2c  x9 = 0x12344800  (addi negativo resta al lui)
lui   x0, 0x12345           # 0x30  x0 sigue en 0
auipc x0, 1                 # 0x34  x0 sigue en 0
ebreak                      # 0x38

#   x5 = 0x00001010   x6 = 0xfffff014   x7 = 0x7fffffff   x8 = 0x12345678
#   x9 = 0x12344800   x0 = 0x00000000
